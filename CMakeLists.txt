cmake_minimum_required(VERSION 3.24)
project(Glancelot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(GLANCELOT_BUILD_VIEWER "Build the Qt viewer" ON)
add_library(glancelot_core src/core/navigation.cpp)
target_include_directories(glancelot_core PUBLIC include)
if(MSVC)
  target_compile_options(glancelot_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(glancelot_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(navigation_tests tests/navigation_tests.cpp)
  target_link_libraries(navigation_tests PRIVATE glancelot_core)
  add_test(NAME navigation COMMAND navigation_tests)
endif()
if(GLANCELOT_BUILD_VIEWER)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 ShaderTools Test Concurrent)
  qt_standard_project_setup(REQUIRES 6.8)
  qt_add_executable(glancelot app/main.cpp src/app/viewer_controller.cpp include/glancelot/viewer_controller.hpp src/render/image_canvas.cpp src/render/image_canvas.hpp)
  target_include_directories(glancelot PRIVATE include src/render)
  target_link_libraries(glancelot PRIVATE glancelot_core Qt6::Quick Qt6::QuickControls2 Qt6::GuiPrivate Qt6::Concurrent)
  qt_add_qml_module(glancelot URI Glancelot VERSION 1.0 QML_FILES ui/Main.qml)
  qt_add_shaders(glancelot "canvas_shaders" PREFIX "/shaders" FILES src/render/image.vert src/render/image.frag)
  qt_add_executable(glancelot-decoder src/worker/main.cpp)
  target_include_directories(glancelot-decoder PRIVATE include)
  target_link_libraries(glancelot-decoder PRIVATE Qt6::Core Qt6::Gui)
  install(TARGETS glancelot glancelot-decoder RUNTIME DESTINATION .)
  qt_generate_deploy_qml_app_script(TARGET glancelot OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
